import { useEffect, useState } from 'react';
import { Link, useNavigate } from 'react-router-dom';
import { DashboardShell } from '../../widgets/dashboard-shell';
import { StatusPill } from '../../shared/ui/StatusPill';
import { Select } from '../../shared/ui/Select';
import { CannedResponsesPanel } from '../../widgets/canned-responses';
import {
    getAccount,
    getAdminTickets,
    ApiError,
    type AccountUser,
    type TicketStatus,
    type TicketSummary,
} from '../../shared/api/statsAuth';
import styles from './index.module.scss';

const CATEGORY_LABELS: Record<string, string> = {
    account: 'Account & login',
    bug: 'Bug report',
    billing: 'Billing',
    other: 'Other',
};

const STATUS_OPTIONS = [
    { value: '', label: 'All' },
    { value: 'open', label: 'Open' },
    { value: 'in_progress', label: 'In progress' },
    { value: 'closed', label: 'Closed' },
];

export function AdminTicketsPage() {
    const navigate = useNavigate();
    const [account, setAccount] = useState<AccountUser | null>(null);
    const [checkingAccount, setCheckingAccount] = useState(true);
    const [status, setStatus] = useState<TicketStatus | ''>('');
    const [tickets, setTickets] = useState<TicketSummary[] | null>(null);
    const [error, setError] = useState<string | null>(null);

    useEffect(() => {
        let cancelled = false;
        getAccount().then(({ user, banned }) => {
            if (cancelled) return;
            if (banned || !user) {
                navigate('/login');
                return;
            }
            setAccount(user);
            setCheckingAccount(false);
        });
        return () => {
            cancelled = true;
        };
    }, [navigate]);

    useEffect(() => {
        if (checkingAccount) return;
        let cancelled = false;
        setTickets(null);
        setError(null);
        getAdminTickets(status || undefined)
            .then(result => {
                if (!cancelled) setTickets(result.tickets);
            })
            .catch(err => {
                if (cancelled) return;
                setError(err instanceof ApiError ? err.message : 'Something went wrong — please try again.');
            });
        return () => {
            cancelled = true;
        };
    }, [checkingAccount, status]);

    const dashUser = account ? { email: account.email, isAdmin: account.isAdmin } : null;

    return (
        <DashboardShell
            user={dashUser}
            title="Support Tickets"
            subtitle={tickets ? `${tickets.length} ticket${tickets.length === 1 ? '' : 's'}` : undefined}
            error={error}
            wide
        >
            {checkingAccount ? (
                <div className={styles.loadingState}>Loading…</div>
            ) : (
                !error && (
                    <div className={styles.pageWrapper}>
                        <div className={styles.headerArea}>
                            <div style={{ width: '200px' }}>
                                <Select
                                    options={STATUS_OPTIONS}
                                    value={status}
                                    onChange={val => setStatus(val as TicketStatus | '')}
                                />
                            </div>
                        </div>

                        <div className={styles.listContainer}>
                            {tickets === null && <p className={styles.empty}>Loading…</p>}
                            {tickets?.length === 0 && <p className={styles.empty}>No tickets found.</p>}
                            {tickets && tickets.length > 0 && (
                                <ul className={styles.list}>
                                    {tickets.map(t => (
                                        <li key={t.number}>
                                            <Link to={`/admin/tickets/${t.number}`} className={styles.row}>
                                                <div className={styles.rowLeft}>
                                                    <StatusPill status={t.status} />
                                                    <span className={styles.number}>#{t.number}</span>
                                                </div>
                                                <div className={styles.rowMain}>
                                                    <div className={styles.subject}>{t.subject}</div>
                                                    <div className={styles.meta}>{t.email}</div>
                                                </div>
                                                <span className={styles.category}>
                                                    {CATEGORY_LABELS[t.category] ?? t.category}
                                                </span>
                                                <span className={styles.time}>{t.time}</span>
                                            </Link>
                                        </li>
                                    ))}
                                </ul>
                            )}
                        </div>

                        <div className={styles.bottomSection}>
                            <CannedResponsesPanel />
                        </div>
                    </div>
                )
            )}
        </DashboardShell>
    );
}

import { useEffect, useState, type FormEvent } from 'react';
import { Link, useNavigate, useParams } from 'react-router-dom';
import { DashboardShell } from '../../widgets/dashboard-shell';
import { Panel } from '../../shared/ui/Panel';
import { StatusPill } from '../../shared/ui/StatusPill';
import { Select } from '../../shared/ui/Select';
import { TextArea } from '../../shared/ui/TextArea';
import { Button } from '../../shared/ui/Button';
import { TicketThread } from '../../widgets/ticket-thread';
import { CannedResponsesPanel } from '../../widgets/canned-responses';
import {
    getAccount,
    getAdminTicket,
    replyToTicketAsAdmin,
    setTicketStatus,
    verifyUser,
    type AccountUser,
    type AdminTicketDetail,
    type TicketReply,
    type TicketStatus,
} from '../../shared/api/statsAuth';
import styles from './index.module.scss';

const STATUS_OPTIONS = [
    { value: 'open', label: 'Open' },
    { value: 'in_progress', label: 'In progress' },
    { value: 'closed', label: 'Closed' },
];

export function AdminTicketPage() {
    const { number } = useParams<{ number: string }>();
    const navigate = useNavigate();
    const [account, setAccount] = useState<AccountUser | null>(null);
    const [checking, setChecking] = useState(true);
    const [ticket, setTicket] = useState<AdminTicketDetail | null>(null);
    const [replies, setReplies] = useState<TicketReply[]>([]);
    const [message, setMessage] = useState('');
    const [pending, setPending] = useState(false);
    const [isVerifying, setIsVerifying] = useState(false);

    useEffect(() => {
        getAccount().then(({ user, banned }) => {
            if (banned) {
                navigate('/login');
                return;
            }
            if (!user) {
                navigate('/login');
            } else {
                setAccount(user);
                setChecking(false);
            }
        });
    }, [navigate]);

    useEffect(() => {
        if (checking || !number) return;
        getAdminTicket(number)
            .then(res => {
                setTicket(res.ticket);
                setReplies(res.replies);
            })
            .catch(() => {});
    }, [checking, number]);

    async function onReply(e: FormEvent) {
        e.preventDefault();
        if (!number || !message.trim()) return;
        setPending(true);
        try {
            await replyToTicketAsAdmin(number, message);
            setMessage('');
            const res = await getAdminTicket(number);
            setTicket(res.ticket);
            setReplies(res.replies);
        } catch {
            alert('Failed to send reply');
        } finally {
            setPending(false);
        }
    }

    async function onStatusChange(newStatus: string) {
        if (!number) return;
        try {
            await setTicketStatus(number, newStatus as TicketStatus);
            setTicket(t => (t ? { ...t, status: newStatus as TicketStatus } : null));
        } catch {
            alert('Failed to update status');
        }
    }

    async function handleVerify() {
        if (!ticket || !ticket.userId) return;
        setIsVerifying(true);
        try {
            const newStatus = !ticket.isUserVerified;
            await verifyUser(ticket.userId, newStatus);
            setTicket(t => (t ? { ...t, isUserVerified: newStatus } : null));
        } catch {
            alert('Failed to update verification status');
        } finally {
            setIsVerifying(false);
        }
    }

    if (checking || !ticket)
        return (
            <DashboardShell user={null} title="Loading...">
                Loading...
            </DashboardShell>
        );

    const threadMessages = [
        { id: 'original', isSupport: false, authorLabel: 'User', message: ticket.message, time: ticket.time },
        ...replies.map(r => ({
            id: r.id,
            isSupport: r.author === 'admin',
            authorLabel: r.author === 'admin' ? 'Support (You)' : 'User',
            message: r.message,
            time: r.time,
        })),
    ];

    return (
        <DashboardShell user={{ email: account!.email, isAdmin: account!.isAdmin }} title="" wide>
            <div className={styles.container}>
                <div className={styles.header}>
                    <div>
                        <Link to="/admin/tickets" className={styles.backBtn}>
                            ← Back to all tickets
                        </Link>
                        <h1 className={styles.title}>{ticket.subject}</h1>
                        <div className={styles.meta}>
                            Ticket #{ticket.number} • {ticket.email}
                        </div>
                    </div>
                    <StatusPill status={ticket.status} />
                </div>

                <div className={styles.layout}>
                    <div className={styles.main}>
                        <Panel>
                            <TicketThread messages={threadMessages} />

                            {ticket.status !== 'closed' ? (
                                <form className={styles.replyForm} onSubmit={onReply}>
                                    <TextArea
                                        id="reply"
                                        label="Reply"
                                        placeholder="Type your reply here..."
                                        required
                                        value={message}
                                        onChange={e => setMessage(e.target.value)}
                                    />
                                    <Button type="submit" variant="flat" disabled={pending || !message.trim()}>
                                        {pending ? 'Sending…' : 'Send reply'}
                                    </Button>
                                </form>
                            ) : (
                                <p className={styles.closedNotice}>This ticket is closed.</p>
                            )}
                        </Panel>
                    </div>

                    <aside className={styles.sidebar}>
                        <Panel title="User Profile">
                            <div className={styles.propertyRow}>
                                <span className={styles.propertyLabel}>Email</span>
                                <span className={styles.propertyValue}>{ticket.email}</span>
                            </div>
                            <div className={styles.propertyRow}>
                                <span className={styles.propertyLabel}>Verified</span>
                                <span className={styles.propertyValue}>{ticket.isUserVerified ? 'Yes' : 'No'}</span>
                            </div>
                            <div style={{ marginTop: '16px' }}>
                                <Button
                                    variant="flat"
                                    fullWidth
                                    onClick={handleVerify}
                                    disabled={isVerifying || !ticket.userId}
                                    style={{
                                        backgroundColor: ticket.isUserVerified ? 'var(--surface-2)' : 'var(--accent)',
                                    }}
                                >
                                    {isVerifying
                                        ? 'Updating...'
                                        : ticket.isUserVerified
                                          ? 'Remove Checkmark'
                                          : 'Give Blue Checkmark'}
                                </Button>
                            </div>
                        </Panel>

                        <Panel title="Properties">
                            <div className={styles.propertyRow}>
                                <span className={styles.propertyLabel}>Status</span>
                                <div style={{ width: '140px' }}>
                                    <Select options={STATUS_OPTIONS} value={ticket.status} onChange={onStatusChange} />
                                </div>
                            </div>
                            <div className={styles.propertyRow}>
                                <span className={styles.propertyLabel}>Category</span>
                                <span className={styles.propertyValue} style={{ textTransform: 'capitalize' }}>
                                    {ticket.category}
                                </span>
                            </div>
                            <div className={styles.propertyRow}>
                                <span className={styles.propertyLabel}>Created</span>
                                <span className={styles.propertyValue}>{ticket.time}</span>
                            </div>
                        </Panel>

                        <CannedResponsesPanel />
                    </aside>
                </div>
            </div>
        </DashboardShell>
    );
}

import { Avatar } from '../../shared/ui/Avatar';
import styles from './index.module.scss';

interface ThreadMessage {
    id: string;
    isSupport: boolean;
    authorLabel: string;
    message: string;
    time: string;
}

interface TicketThreadProps {
    messages: ThreadMessage[];
}

// Общий компонент треда для /admin/tickets/:id и /support/ticket/:token —
// оба раньше дублировали одну и ту же вёрстку "пузырей" с небольшими
// расхождениями в подписях (Support/You/email). Собирать итоговый список
// ThreadMessage — забота каждой страницы (она знает, кто смотрит), сам
// компонент просто рисует чат с аватаркой-инициалами на сообщение, как в
// обычных тикет-системах, а не абзацы подряд без разделения по автору.
export function TicketThread({ messages }: TicketThreadProps) {
    return (
        <div className={styles.thread}>
            {messages.map(m => (
                <div key={m.id} className={styles.message}>
                    <Avatar label={m.authorLabel} isSupport={m.isSupport} />
                    <div className={styles.bubbleCol}>
                        <div className={styles.bubbleHead}>
                            <span className={styles.author}>{m.authorLabel}</span>
                            <span className={styles.time}>{m.time}</span>
                        </div>
                        <div className={`${styles.bubble} ${m.isSupport ? styles.bubbleSupport : ''}`}>{m.message}</div>
                    </div>
                </div>
            ))}
        </div>
    );
}

import { useEffect, useState, type FormEvent } from 'react';
import { Panel } from '../../shared/ui/Panel';
import { TextField } from '../../shared/ui/TextField';
import { TextArea } from '../../shared/ui/TextArea';
import { Button } from '../../shared/ui/Button';
import {
    getCannedResponses,
    createCannedResponse,
    deleteCannedResponse,
    ApiError,
    type CannedResponse,
} from '../../shared/api/statsAuth';
import styles from './index.module.scss';

export function CannedResponsesPanel() {
    const [responses, setResponses] = useState<CannedResponse[] | null>(null);
    const [title, setTitle] = useState('');
    const [body, setBody] = useState('');
    const [error, setError] = useState<string | null>(null);
    const [pending, setPending] = useState(false);

    function load() {
        getCannedResponses()
            .then(result => setResponses(result.responses))
            .catch(() => setResponses([]));
    }

    useEffect(load, []);

    async function onAdd(e: FormEvent) {
        e.preventDefault();
        setError(null);
        setPending(true);
        try {
            await createCannedResponse(title, body);
            setTitle('');
            setBody('');
            load();
        } catch (err) {
            setError(err instanceof ApiError ? err.message : 'Something went wrong — please try again.');
        } finally {
            setPending(false);
        }
    }

    async function onDelete(id: string) {
        await deleteCannedResponse(id);
        load();
    }

    return (
        <Panel title="Canned responses" subtitle="Reusable reply templates, available when replying to a ticket.">
            {error && (
                <p className={styles.error} role="alert">
                    {error}
                </p>
            )}

            {responses && responses.length > 0 && (
                <ul className={styles.list}>
                    {responses.map(r => (
                        <li key={r.id} className={styles.item}>
                            <div className={styles.itemTitle}>{r.title}</div>
                            <div className={styles.itemBody}>{r.body}</div>
                            <button type="button" className={styles.deleteBtn} onClick={() => onDelete(r.id)}>
                                Delete
                            </button>
                        </li>
                    ))}
                </ul>
            )}

            <form className={styles.form} onSubmit={onAdd}>
                <TextField
                    id="canned-title"
                    label="Title"
                    required
                    value={title}
                    onChange={e => setTitle(e.target.value)}
                />
                <TextArea id="canned-body" label="Body" required value={body} onChange={e => setBody(e.target.value)} />
                <Button type="submit" variant="ghost" disabled={pending || !title.trim() || !body.trim()}>
                    {pending ? 'Adding…' : 'Add canned response'}
                </Button>
            </form>
        </Panel>
    );
}

import type { AnchorHTMLAttributes, ButtonHTMLAttributes, ReactNode } from 'react';
import styles from './index.module.scss';

type Variant = 'primary' | 'ghost' | 'flat';
type Size = 'md' | 'sm';

interface CommonProps {
    variant?: Variant;
    size?: Size;
    fullWidth?: boolean;
    children: ReactNode;
}

type ButtonProps = CommonProps &
    ButtonHTMLAttributes<HTMLButtonElement> & {
        href?: undefined;
    };

type LinkProps = CommonProps &
    AnchorHTMLAttributes<HTMLAnchorElement> & {
        href: string;
    };

// Кнопка-ссылка (OAuth-редирект на /auth/riot/login) или обычная <button> —
// один и тот же паттерн, что в frontend/shared/ui/Button: разная семантика
// тега, одна визуальная оболочка, вместо двух почти одинаковых компонентов.
export function Button({
    variant = 'primary',
    size = 'md',
    fullWidth,
    children,
    className,
    href,
    ...rest
}: ButtonProps | LinkProps) {
    const classes = [styles.base, styles[variant], styles[size], fullWidth ? styles.fullWidth : '', className]
        .filter(Boolean)
        .join(' ');

    if (href !== undefined) {
        return (
            <a className={classes} href={href} {...(rest as AnchorHTMLAttributes<HTMLAnchorElement>)}>
                {children}
            </a>
        );
    }

    return (
        <button className={classes} {...(rest as ButtonHTMLAttributes<HTMLButtonElement>)}>
            {children}
        </button>
    );
}

import { useState, useRef, useEffect, type KeyboardEvent } from 'react';
import styles from './index.module.scss';

interface Option {
    value: string;
    label: string;
}

interface SelectProps {
    options: Option[];
    value: string;
    onChange: (value: string) => void;
    placeholder?: string;
}

export function Select({ options, value, onChange, placeholder = 'Select...' }: SelectProps) {
    const [isOpen, setIsOpen] = useState(false);
    const containerRef = useRef<HTMLDivElement>(null);

    const selectedOption = options.find(o => o.value === value);

    // Закрываем при клике вне компонента
    useEffect(() => {
        function handleClickOutside(event: MouseEvent) {
            if (containerRef.current && !containerRef.current.contains(event.target as Node)) {
                setIsOpen(false);
            }
        }
        document.addEventListener('mousedown', handleClickOutside);
        return () => document.removeEventListener('mousedown', handleClickOutside);
    }, []);

    const handleKeyDown = (e: KeyboardEvent<HTMLDivElement>) => {
        if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault();
            setIsOpen(!isOpen);
        } else if (e.key === 'Escape') {
            setIsOpen(false);
        }
    };

    return (
        <div className={styles.container} ref={containerRef}>
            <div
                className={`${styles.trigger} ${isOpen ? styles.active : ''}`}
                onClick={() => setIsOpen(!isOpen)}
                onKeyDown={handleKeyDown}
                tabIndex={0}
                role="button"
                aria-haspopup="listbox"
                aria-expanded={isOpen}
            >
                <span>{selectedOption ? selectedOption.label : placeholder}</span>
                <svg
                    className={`${styles.icon} ${isOpen ? styles.open : ''}`}
                    viewBox="0 0 24 24"
                    fill="none"
                    stroke="currentColor"
                    strokeWidth="2"
                    strokeLinecap="round"
                    strokeLinejoin="round"
                >
                    <polyline points="6 9 12 15 18 9"></polyline>
                </svg>
            </div>

            {isOpen && (
                <div className={styles.menu} role="listbox">
                    {options.map(option => (
                        <div
                            key={option.value}
                            className={`${styles.option} ${option.value === value ? styles.selected : ''}`}
                            onClick={() => {
                                onChange(option.value);
                                setIsOpen(false);
                            }}
                            role="option"
                            aria-selected={option.value === value}
                        >
                            {option.label}
                            {option.value === value && (
                                <svg
                                    className={styles.check}
                                    viewBox="0 0 24 24"
                                    fill="none"
                                    stroke="currentColor"
                                    strokeWidth="3"
                                    strokeLinecap="round"
                                    strokeLinejoin="round"
                                >
                                    <polyline points="20 6 9 17 4 12"></polyline>
                                </svg>
                            )}
                        </div>
                    ))}
                </div>
            )}
        </div>
    );
}

async function postJson<T>(path: string, body: unknown): Promise<T> {
    const res = await fetch(path, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        credentials: 'same-origin',
        body: JSON.stringify(body),
    });
    const data = await res.json().catch(() => ({}));
    if (!res.ok) {
        const code = typeof data.error === 'string' ? data.error : 'internal_error';
        throw new ApiError(code, messageFor(code));
    }
    return data as T;
}

async function getJson<T>(path: string): Promise<T> {
    const res = await fetch(path, { credentials: 'same-origin' });
    const data = await res.json().catch(() => ({}));
    if (!res.ok) {
        const code = typeof data.error === 'string' ? data.error : 'internal_error';
        throw new ApiError(code, messageFor(code));
    }
    return data as T;
}

package tech.extree.stats.adapter.in.web;

import jakarta.servlet.http.HttpServletRequest;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.CookieValue;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;
import tech.extree.stats.adapter.in.web.dto.CannedResponsesResponse;
import tech.extree.stats.adapter.in.web.dto.CreateCannedResponseRequest;
import tech.extree.stats.adapter.in.web.dto.CreateCannedResponseResponse;
import tech.extree.stats.adapter.in.web.dto.ErrorResponse;
import tech.extree.stats.adapter.in.web.dto.MessageRequest;
import tech.extree.stats.adapter.in.web.dto.OkResponse;
import tech.extree.stats.adapter.in.web.dto.SetTicketStatusRequest;
import tech.extree.stats.adapter.in.web.dto.TicketDetailResponse;
import tech.extree.stats.adapter.in.web.dto.TicketListResponse;
import tech.extree.stats.application.port.in.AdminTicketUseCase;

/**
 * Соответствует requireAdmin + /api/admin/tickets*, /api/admin/canned-responses*
 * в server.js — вся админка тикет-платформы.
 */
@RestController
public class AdminTicketController {

    private final AdminTicketUseCase adminTicketUseCase;

    public AdminTicketController(AdminTicketUseCase adminTicketUseCase) {
        this.adminTicketUseCase = adminTicketUseCase;
    }

    @GetMapping("/api/admin/tickets")
    public ResponseEntity<?> listTickets(
            @RequestParam(required = false) String status,
            @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie) {
        AdminTicketUseCase.ListResult result = adminTicketUseCase.listTickets(cookie, status);
        return switch (result.outcome()) {
            case OK -> ResponseEntity.ok(new TicketListResponse(result.tickets()));
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
            case INVALID_REQUEST -> badRequest("invalid_request");
        };
    }

    @GetMapping("/api/admin/tickets/{number}")
    public ResponseEntity<?> getTicket(
            @PathVariable String number, @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie) {
        AdminTicketUseCase.DetailResult result = adminTicketUseCase.getTicket(cookie, number);
        return switch (result.outcome()) {
            case OK -> ResponseEntity.ok(new TicketDetailResponse(result.ticket(), result.replies()));
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
            case TICKET_NOT_FOUND -> status(HttpStatus.NOT_FOUND, "ticket_not_found");
        };
    }

    @PostMapping("/api/admin/tickets/{number}/reply")
    public ResponseEntity<?> reply(
            @PathVariable String number,
            @RequestBody MessageRequest request,
            @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie,
            HttpServletRequest httpRequest) {
        AdminTicketUseCase.ReplyOutcome outcome =
                adminTicketUseCase.reply(cookie, number, request.message(), httpRequest.getHeader(HttpHeaders.HOST));
        return switch (outcome) {
            case OK -> ResponseEntity.ok(OkResponse.OK);
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
            case INVALID_MESSAGE -> badRequest("invalid_message");
            case TICKET_NOT_FOUND -> status(HttpStatus.NOT_FOUND, "ticket_not_found");
        };
    }

    @PostMapping("/api/admin/tickets/{number}/status")
    public ResponseEntity<?> setStatus(
            @PathVariable String number,
            @RequestBody SetTicketStatusRequest request,
            @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie) {
        AdminTicketUseCase.StatusOutcome outcome = adminTicketUseCase.setStatus(cookie, number, request.status());
        return switch (outcome) {
            case OK -> ResponseEntity.ok(OkResponse.OK);
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
            case INVALID_REQUEST -> badRequest("invalid_request");
            case TICKET_NOT_FOUND -> status(HttpStatus.NOT_FOUND, "ticket_not_found");
        };
    }

    @GetMapping("/api/admin/canned-responses")
    public ResponseEntity<?> listCannedResponses(
            @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie) {
        AdminTicketUseCase.CannedListResult result = adminTicketUseCase.listCannedResponses(cookie);
        return switch (result.outcome()) {
            case OK -> ResponseEntity.ok(new CannedResponsesResponse(result.responses()));
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
        };
    }

    @PostMapping("/api/admin/canned-responses")
    public ResponseEntity<?> createCannedResponse(
            @RequestBody CreateCannedResponseRequest request,
            @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie) {
        AdminTicketUseCase.CannedCreateResult result =
                adminTicketUseCase.createCannedResponse(cookie, request.title(), request.body());
        return switch (result.outcome()) {
            case OK -> ResponseEntity.ok(new CreateCannedResponseResponse(true, result.id()));
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
            case INVALID_SUBJECT -> badRequest("invalid_subject");
            case INVALID_MESSAGE -> badRequest("invalid_message");
        };
    }

    @DeleteMapping("/api/admin/canned-responses/{id}")
    public ResponseEntity<?> deleteCannedResponse(
            @PathVariable String id, @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie) {
        AdminTicketUseCase.CannedDeleteOutcome outcome = adminTicketUseCase.deleteCannedResponse(cookie, id);
        return switch (outcome) {
            case OK -> ResponseEntity.ok(OkResponse.OK);
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
        };
    }

    private static ResponseEntity<ErrorResponse> badRequest(String error) {
        return status(HttpStatus.BAD_REQUEST, error);
    }

    private static ResponseEntity<ErrorResponse> status(HttpStatus status, String error) {
        return ResponseEntity.status(status).body(new ErrorResponse(error));
    }
}

package tech.extree.stats.application.port.in;

import java.util.List;
import java.util.Optional;

/**
 * Соответствует POST /api/support/tickets, GET /api/support/tickets/:token,
 * POST /api/support/tickets/:token/reply в server.js — публичная сторона
 * тикет-платформы (без входа или с сессией, но без прав админа).
 */
public interface SupportTicketUseCase {

    record ReplyView(String id, String author, String message, String time) {}

    enum CreateOutcome {
        OK,
        INVALID_REQUEST,
        INVALID_EMAIL,
        INVALID_SUBJECT,
        INVALID_MESSAGE,
        INVALID_CATEGORY,
        CODE_ON_COOLDOWN,
        INTERNAL_ERROR
    }

    record CreateResult(CreateOutcome outcome, Long ticketNumber) {
        public static CreateResult error(CreateOutcome outcome) {
            return new CreateResult(outcome, null);
        }
    }

    /**
     * Если cookieValue резолвится в действующую сессию — email аккаунта
     * используется и rawBodyEmail игнорируется (не даём залогиненному
     * притвориться другим отправителем), как и Node. requestHost — для
     * ссылки в письме-подтверждении.
     */
    CreateResult createTicket(
            String cookieValue, String rawBodyEmail, String subject, String message, String category, String requestHost);

    record PublicTicketView(long number, String subject, String message, String status, String time) {}

    record PublicTicketResult(PublicTicketView ticket, List<ReplyView> replies) {}

    Optional<PublicTicketResult> getTicketByToken(String token);

    enum ReplyOutcome {
        OK,
        TICKET_NOT_FOUND,
        TICKET_CLOSED,
        INVALID_MESSAGE,
        CODE_ON_COOLDOWN,
        INTERNAL_ERROR
    }

    ReplyOutcome replyAsUser(String token, String message, String requestHost);
}
