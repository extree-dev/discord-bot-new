package tech.extree.stats.adapter.in.web;

import jakarta.servlet.http.HttpServletRequest;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import tech.extree.stats.adapter.in.web.dto.AdminUserResponse;
import tech.extree.stats.adapter.in.web.dto.OkResponse;
import tech.extree.stats.application.port.out.AdminAuditRepository;
import tech.extree.stats.application.port.out.SessionCodec;
import tech.extree.stats.application.port.out.UserRepository;
import tech.extree.stats.domain.Session;
import tech.extree.stats.domain.User;

import java.util.List;
import java.util.Optional;

@RestController
public class AdminUserController {

    private final UserRepository userRepository;
    private final SessionCodec sessionCodec;
    private final AdminAuditRepository auditRepository;

    public AdminUserController(UserRepository userRepository, SessionCodec sessionCodec, AdminAuditRepository auditRepository) {
        this.userRepository = userRepository;
        this.sessionCodec = sessionCodec;
        this.auditRepository = auditRepository;
    }

    private boolean isAdmin(String cookie) {
        if (cookie == null) return false;
        Optional<Session> session = sessionCodec.decrypt(cookie);
        if (session.isEmpty() || session.get().userId() == null) return false;
        return userRepository.findById(session.get().userId())
                .map(User::isAdmin)
                .orElse(false);
    }

    @GetMapping("/api/admin/users")
    public ResponseEntity<?> listUsers(@CookieValue(name = "extree_stats_session", required = false) String cookie) {
        if (!isAdmin(cookie)) return ResponseEntity.status(403).body(new OkResponse(false));

        List<AdminUserResponse.AdminUser> users = userRepository.findAllUsers().stream()
                .map(u -> new AdminUserResponse.AdminUser(
                        u.id(), u.email(), u.name(), u.nickname(), u.avatarUrl(),
                        u.isVerified(), u.isBanned(), u.isAdmin()))
                .toList();

        return ResponseEntity.ok(new AdminUserResponse(users));
    }

    @PostMapping("/api/admin/users/{userId}/verify")
    public ResponseEntity<?> verifyUser(
            @PathVariable String userId,
            @CookieValue(name = "extree_stats_session", required = false) String cookie,
            @RequestParam boolean isVerified, HttpServletRequest httpRequest) {
        if (!isAdmin(cookie)) return ResponseEntity.status(403).body(new OkResponse(false));
        userRepository.setVerified(userId, isVerified);
        String adminId = currentAdminId(cookie);
        if (adminId != null) {
            auditRepository.record(adminId, "user.verify", "user", userId,
                    "{\"isVerified\":" + isVerified + "}", ClientIp.from(httpRequest));
        }
        return ResponseEntity.ok(OkResponse.OK);
    }

    @PostMapping("/api/admin/users/{userId}/ban")
    public ResponseEntity<?> banUser(
            @PathVariable String userId,
            @CookieValue(name = "extree_stats_session", required = false) String cookie,
            @RequestParam boolean isBanned, HttpServletRequest httpRequest) {
        if (!isAdmin(cookie)) return ResponseEntity.status(403).body(new OkResponse(false));
        userRepository.setBanned(userId, isBanned);
        String adminId = currentAdminId(cookie);
        if (adminId != null) {
            auditRepository.record(adminId, "user.ban", "user", userId,
                    "{\"isBanned\":" + isBanned + "}", ClientIp.from(httpRequest));
        }
        return ResponseEntity.ok(OkResponse.OK);
    }

    // НОВЫЙ ЭНДПОИНТ: Назначить/снять админа
    @PostMapping("/api/admin/users/{userId}/make-admin")
    public ResponseEntity<?> makeAdmin(
            @PathVariable String userId,
            @CookieValue(name = "extree_stats_session", required = false) String cookie,
            @RequestParam boolean isAdmin, HttpServletRequest httpRequest) {
        if (!isAdmin(cookie)) return ResponseEntity.status(403).body(new OkResponse(false));
        userRepository.setAdmin(userId, isAdmin);
        String adminId = currentAdminId(cookie);
        if (adminId != null) {
            auditRepository.record(adminId, "user.make_admin", "user", userId,
                    "{\"isAdmin\":" + isAdmin + "}", ClientIp.from(httpRequest));
        }
        return ResponseEntity.ok(OkResponse.OK);
    }

    private String currentAdminId(String cookie) {
        if (cookie == null) return null;
        return sessionCodec.decrypt(cookie)
                .filter(s -> s.userId() != null)
                .map(s -> s.userId())
                .orElse(null);
    }
}

@RestController
public class AdminTicketController {

    private final AdminTicketUseCase adminTicketUseCase;

    public AdminTicketController(AdminTicketUseCase adminTicketUseCase) {
        this.adminTicketUseCase = adminTicketUseCase;
    }

    @GetMapping("/api/admin/tickets")
    public ResponseEntity<?> listTickets(
            @RequestParam(required = false) String status,
            @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie) {
        AdminTicketUseCase.ListResult result = adminTicketUseCase.listTickets(cookie, status);
        return switch (result.outcome()) {
            case OK -> ResponseEntity.ok(new TicketListResponse(result.tickets()));
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
            case INVALID_REQUEST -> badRequest("invalid_request");
        };
    }

    @GetMapping("/api/admin/tickets/{number}")
    public ResponseEntity<?> getTicket(
            @PathVariable String number, @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie) {
        AdminTicketUseCase.DetailResult result = adminTicketUseCase.getTicket(cookie, number);
        return switch (result.outcome()) {
            case OK -> ResponseEntity.ok(new TicketDetailResponse(result.ticket(), result.replies()));
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
            case TICKET_NOT_FOUND -> status(HttpStatus.NOT_FOUND, "ticket_not_found");
        };
    }

    @PostMapping("/api/admin/tickets/{number}/reply")
    public ResponseEntity<?> reply(
            @PathVariable String number,
            @RequestBody MessageRequest request,
            @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie,
            HttpServletRequest httpRequest) {
        AdminTicketUseCase.ReplyOutcome outcome =
                adminTicketUseCase.reply(cookie, number, request.message(), httpRequest.getHeader(HttpHeaders.HOST));
        return switch (outcome) {
            case OK -> ResponseEntity.ok(OkResponse.OK);
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
            case INVALID_MESSAGE -> badRequest("invalid_message");
            case TICKET_NOT_FOUND -> status(HttpStatus.NOT_FOUND, "ticket_not_found");
        };
    }

    @PostMapping("/api/admin/tickets/{number}/status")
    public ResponseEntity<?> setStatus(
            @PathVariable String number,
            @RequestBody SetTicketStatusRequest request,
            @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie) {
        AdminTicketUseCase.StatusOutcome outcome = adminTicketUseCase.setStatus(cookie, number, request.status());
        return switch (outcome) {
            case OK -> ResponseEntity.ok(OkResponse.OK);
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
            case INVALID_REQUEST -> badRequest("invalid_request");
            case TICKET_NOT_FOUND -> status(HttpStatus.NOT_FOUND, "ticket_not_found");
        };

            case OK -> ResponseEntity.ok(new CannedResponsesResponse(result.responses()));
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");

            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
            case INVALID_SUBJECT -> badRequest("invalid_subject");
            case INVALID_MESSAGE -> badRequest("invalid_message");

        AdminTicketUseCase.CannedDeleteOutcome outcome = adminTicketUseCase.deleteCannedResponse(cookie, id);
        return switch (outcome) {
            case OK -> ResponseEntity.ok(OkResponse.OK);
            case UNAUTHORIZED -> status(HttpStatus.UNAUTHORIZED, "unauthorized");
            case FORBIDDEN -> status(HttpStatus.FORBIDDEN, "forbidden");
        };
    }

    private static ResponseEntity<ErrorResponse> badRequest(String error) {
        return status(HttpStatus.BAD_REQUEST, error);
    }

    private static ResponseEntity<ErrorResponse> status(HttpStatus status, String error) {
        return ResponseEntity.status(status).body(new ErrorResponse(error));
    }

interface SiteHeaderProps {

    actions?: ReactNode;
}

// SRP: только хедер. Если передан явный rightSlot/actions — рендерим их
// (как раньше). Если НЕ переданы — рендерим дефолтные по статусу auth:
// - authenticated → Dashboard + ProfileDropdown
// - anonymous → Log in / Sign up
export function SiteHeader({ rightSlot, actions }: SiteHeaderProps) {
    const { status, user } = useAuth();

    const showDefaultActions = rightSlot === undefined && actions === undefined;

    return (
        <header className={styles.header}>
            <Link to="/" className={styles.brandLink}>
                <img src="/logo.png" alt="" className={styles.mark} aria-hidden="true" />
                <span className={styles.brand}>Extree Stats</span>
            </Link>
            <div className={styles.navRight}>
                {rightSlot && <div className={styles.right}>{rightSlot}</div>}
                {showDefaultActions ? (
                    <div className={styles.actions}>
                        {status === 'authenticated' && user ? (
                            <>
                                <Button href="/account" variant="primary" size="sm">
                                    Dashboard
                                </Button>
                                <ProfileDropdown email={user.email} isAdmin={user.isAdmin} />
                            </>
                        ) : (
                            <>
                                <Button href="/login" variant="ghost" size="sm">
                                    Log in
                                </Button>
                                <Button href="/register" variant="primary" size="sm">
                                    Sign up
                                </Button>
                            </>
                        )}
                    </div>
                ) : (
                    actions && <div className={styles.actions}>{actions}</div>
                )}
            </div>
        </header>
    );
}

import { useState, useRef, useEffect } from 'react';
import { Link } from 'react-router-dom';
import { TicketIcon, SettingsIcon, ShieldIcon, PowerIcon, UsersIcon } from './icons';
import styles from './index.module.scss';

interface ProfileDropdownProps {
    email: string;
    isAdmin?: boolean;
}

export function ProfileDropdown({ email, isAdmin }: ProfileDropdownProps) {
    const [isOpen, setIsOpen] = useState(false);
    const menuRef = useRef<HTMLDivElement>(null);

    useEffect(() => {
        function handleClickOutside(event: MouseEvent) {
            if (menuRef.current && !menuRef.current.contains(event.target as Node)) {
                setIsOpen(false);
            }
        }
        document.addEventListener('mousedown', handleClickOutside);
        return () => document.removeEventListener('mousedown', handleClickOutside);
    }, []);

    const handleSignOut = () => {
        window.location.href = '/auth/logout';
    };

    const initial = email ? email[0].toUpperCase() : '?';

    return (
        <div className={styles.container} ref={menuRef}>
            <button className={`${styles.avatarBtn} ${isOpen ? styles.active : ''}`} onClick={() => setIsOpen(!isOpen)}>
                {initial}
            </button>

            {isOpen && (
                <div className={styles.menu}>
                    <div className={styles.menuHeader}>
                        <span className={styles.brandName}>Extree Stats</span>
                    </div>

                    <div className={styles.menuList}>
                        <Link to="/account/settings" className={styles.menuItem} onClick={() => setIsOpen(false)}>
                            <SettingsIcon /> Account Settings
                        </Link>

                        <Link to="/support" className={styles.menuItem} onClick={() => setIsOpen(false)}>
                            <TicketIcon /> Support
                        </Link>

                        {isAdmin && (
                            <>
                                <Link to="/admin/tickets" className={styles.menuItem} onClick={() => setIsOpen(false)}>
                                    <ShieldIcon /> Admin Tickets
                                </Link>
                                <Link to="/admin/users" className={styles.menuItem} onClick={() => setIsOpen(false)}>
                                    <UsersIcon /> Manage Users
                                </Link>
                                <Link to="/admin/audit" className={styles.menuItem} onClick={() => setIsOpen(false)}>
                                    <ShieldIcon /> Audit log
                                </Link>
                            </>
                        )}
                    </div>

                    <div className={styles.divider} />

                    <div className={styles.menuList}>
                        <button onClick={handleSignOut} className={styles.menuItem}>
                            <PowerIcon /> Logout
                        </button>
                    </div>
                </div>
            )}
        </div>
    );
}

let csrfTokenCache: string | null = null;
let csrfFetchPromise: Promise<string> | null = null;

/** Один раз тянем /api/csrf-token, дальше кешируем в памяти. */
async function getCsrfToken(): Promise<string> {
    if (csrfTokenCache) return csrfTokenCache;
    if (!csrfFetchPromise) {
        csrfFetchPromise = (async () => {
            const res = await fetch('/api/csrf-token', { credentials: 'same-origin' });
            const data = await res.json().catch(() => ({}));
            const token = typeof data.csrfToken === 'string' ? data.csrfToken : '';
            if (token) csrfTokenCache = token;
            return token;
        })().finally(() => {
            csrfFetchPromise = null;
        });
    }
    return csrfFetchPromise!;
}

/** Сбросить кеш (при 403 csrf_failed — перезапросить). */
function resetCsrfToken() {
    csrfTokenCache = null;
}

async function postJson<T>(path: string, body: unknown): Promise<T> {
    const csrf = await getCsrfToken();
    let res = await fetch(path, {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            ...(csrf ? { 'X-CSRF-Token': csrf } : {}),
        },
        credentials: 'same-origin',
        body: JSON.stringify(body),
    });
    if (res.status === 403) {
        const first = await res
            .clone()
            .json()
            .catch(() => ({}));
        if (first.error === 'csrf_failed') {
            resetCsrfToken();
            const fresh = await getCsrfToken();
            res = await fetch(path, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    ...(fresh ? { 'X-CSRF-Token': fresh } : {}),
                },
                credentials: 'same-origin',
                body: JSON.stringify(body),
            });
        }
    }
    const data = await res.json().catch(() => ({}));
    if (!res.ok) {
        const code = typeof data.error === 'string' ? data.error : 'internal_error';
        throw new ApiError(code, messageFor(code));
    }
    return data as T;
}

    if (!res.ok) {
        const code = typeof data.error === 'string' ? data.error : 'internal_error';
        throw new ApiError(code, messageFor(code));
    }
    return data as T;
}

@RestController
public class TwoFactorController {

    private final SessionCodec sessionCodec;
    private final UserRepository userRepository;
    private final TotpService totpService;
    private final PasswordHasher passwordHasher;

    public TwoFactorController(
            SessionCodec sessionCodec,
            UserRepository userRepository,
            TotpService totpService,
            PasswordHasher passwordHasher) {
        this.sessionCodec = sessionCodec;
        this.userRepository = userRepository;
        this.totpService = totpService;
        this.passwordHasher = passwordHasher;

        if (cookie == null) return Optional.empty();
        Optional<Session> session = sessionCodec.decrypt(cookie).filter(s -> s.userId() != null);

    public ResponseEntity<?> setup(
            @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie) {
        Optional<User> user = currentUser(cookie);
        if (user.isEmpty()) return status(HttpStatus.UNAUTHORIZED, "unauthorized");
        if (user.get().isBanned()) return status(HttpStatus.FORBIDDEN, "banned");
        if (totpService.isEnabled(user.get().id())) {
            return badRequest("totp_required"); // уже включена — сюда не ходим
        }
        TotpService.SetupResult result = totpService.setup(user.get().id(), user.get().email());
        return ResponseEntity.ok(Map.of(
                "secret", result.secret(),
                "otpauthUrl", result.otpauthUrl(),
                "recoveryCodes", result.recoveryCodes()));
    }

    public ResponseEntity<?> enable(
            @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie,
            @RequestBody Map<String, String> body) {
        Optional<User> user = currentUser(cookie);
        if (user.isEmpty()) return status(HttpStatus.UNAUTHORIZED, "unauthorized");
        String code = body == null ? null : body.get("code");
        if (code == null || code.isBlank()) return badRequest("invalid_request");
        if (!totpService.enable(user.get().id(), code.trim())) {
            return status(HttpStatus.UNAUTHORIZED, "totp_invalid");
        }
        return ResponseEntity.ok(OkResponse.OK);
    }

    @PostMapping("/api/account/2fa/disable")
    public ResponseEntity<?> disable(
            @CookieValue(name = SessionCookies.SESSION_COOKIE, required = false) String cookie,

        return ResponseEntity.ok(OkResponse.OK);

        }

        return ResponseEntity.ok(Map.of("recoveryCodes", codes));

package tech.extree.stats.application.service;

import com.warrenstrange.googleauth.GoogleAuthenticator;
import com.warrenstrange.googleauth.GoogleAuthenticatorConfig;
import com.warrenstrange.googleauth.GoogleAuthenticatorKey;
import java.security.MessageDigest;
import java.security.SecureRandom;
import java.time.Instant;
import java.util.ArrayList;
import java.util.List;
import java.util.Locale;
import java.util.Optional;
import org.springframework.stereotype.Service;
import tech.extree.stats.application.port.out.TotpRepository;
import tech.extree.stats.domain.TotpSecret;

/**
 * TOTP 2FA: генерация секрета, otpauth:// URL, проверка кода, recovery-коды.
 *
 * GoogleAuthenticator — стандартная библиотека, совместима с
 * Google Authenticator / Authy / 1Password / Bitwarden (RFC 6238).
 * Recovery-коды хешируются SHA-256 и хранятся в JSONB.
 */
@Service
public class TotpService {

    private static final int RECOVERY_CODE_COUNT = 10;
    private static final SecureRandom SECURE_RANDOM = new SecureRandom();

    private final TotpRepository totpRepository;
    private final GoogleAuthenticator authenticator;

    public TotpService(TotpRepository totpRepository) {
        this.totpRepository = totpRepository;
        GoogleAuthenticatorConfig config = new GoogleAuthenticatorConfig.GoogleAuthenticatorConfigBuilder()
                .setWindowSize(1)  // ±30 сек
                .build();
        this.authenticator = new GoogleAuthenticator(config);
    }

    public record SetupResult(String secret, String otpauthUrl, List<String> recoveryCodes) {}

    /** Шаг 1: сгенерировать секрет + recovery-коды, сохранить pending. */
    public SetupResult setup(String userId, String email) {
        GoogleAuthenticatorKey key = authenticator.createCredentials();
        String secret = key.getKey();
        List<String> plainCodes = new ArrayList<>();
        List<String> hashedCodes = new ArrayList<>();
        for (int i = 0; i < RECOVERY_CODE_COUNT; i++) {
            String code = randomRecoveryCode();
            plainCodes.add(code);
            hashedCodes.add(sha256(code));
        }
        totpRepository.upsertPending(userId, secret, hashedCodes);

        String otpauth = String.format(
                "otpauth://totp/Extree%%20Stats:%s?secret=%s&issuer=Extree%%20Stats&algorithm=SHA1&digits=6&period=30",
                urlEncode(email), secret);

        return new SetupResult(secret, otpauth, plainCodes);
    }

    /** Шаг 2: проверить первый код и активировать. */
    public boolean enable(String userId, String code) {
        Optional<TotpSecret> maybe = totpRepository.findByUserId(userId);
        if (maybe.isEmpty() || maybe.get().isEnabled()) return false;
        if (!verifyCode(maybe.get().secret(), code)) return false;
        totpRepository.enable(userId, Instant.now());
        return true;
    }

    /** Шаг 3: отключить (требует пароль — на уровне контроллера). */
    public void disable(String userId) {
        totpRepository.delete(userId);
    }

    public boolean isEnabled(String userId) {
        return totpRepository.findByUserId(userId).map(TotpSecret::isEnabled).orElse(false);
    }

    /** Проверка 6-значного TOTP-кода при логине. */
    public boolean verifyCode(String secret, String code) {
        if (code == null) return false;
        try {
            int intCode = Integer.parseInt(code.trim());
            return authenticator.authorize(secret, intCode);
        } catch (NumberFormatException e) {
            return false;
        }
    }

/** Проверка 6-значного TOTP-кода по userId (удобно из LoginService). */
    public boolean verifyCodeForUser(String userId, String code) {
        Optional<TotpSecret> maybe = totpRepository.findByUserId(userId);
        if (maybe.isEmpty() || !maybe.get().isEnabled()) return false;
        return verifyCode(maybe.get().secret(), code);
    }

    /** Проверка recovery-кода (одноразовый). Если подошёл — удаляем из БД. */
    public boolean consumeRecoveryCode(String userId, String code) {
        Optional<TotpSecret> maybe = totpRepository.findByUserId(userId);
        if (maybe.isEmpty()) return false;
        String hashed = sha256(code.trim().toLowerCase(Locale.ROOT));
        if (!maybe.get().recoveryCodesHashed().contains(hashed)) return false;
        totpRepository.removeRecoveryCode(userId, hashed);
        return true;
    }

    public List<String> regenerateRecoveryCodes(String userId) {
        List<String> plain = new ArrayList<>();
        List<String> hashed = new ArrayList<>();
        for (int i = 0; i < RECOVERY_CODE_COUNT; i++) {
            String c = randomRecoveryCode();
            plain.add(c);
            hashed.add(sha256(c));
        }
        totpRepository.updateRecoveryCodes(userId, hashed);
        return plain;
    }

    private static String randomRecoveryCode() {
        byte[] b = new byte[5];
        SECURE_RANDOM.nextBytes(b);
        StringBuilder sb = new StringBuilder();
        for (byte x : b) sb.append(String.format("%02x", x));
        // 10 hex-символов → xxxxx-xxxxx
        String s = sb.toString();
        return s.substring(0, 5) + "-" + s.substring(5, 10);
    }

    private static String sha256(String s) {
        try {
            MessageDigest md = MessageDigest.getInstance("SHA-256");
            byte[] digest = md.digest(s.getBytes(java.nio.charset.StandardCharsets.UTF_8));
            StringBuilder sb = new StringBuilder();
            for (byte b : digest) sb.append(String.format("%02x", b));
            return sb.toString();
        } catch (Exception e) {
            throw new IllegalStateException("SHA-256 недоступен", e);
        }
    }

    private static String urlEncode(String s) {
        return java.net.URLEncoder.encode(s, java.nio.charset.StandardCharsets.UTF_8);
    }
}

package tech.extree.stats.adapter.out.persistence;

import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.sql.Timestamp;
import java.time.Instant;
import java.util.List;
import java.util.Optional;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.core.RowMapper;
import org.springframework.stereotype.Repository;
import tech.extree.stats.application.port.out.TotpRepository;
import tech.extree.stats.domain.TotpSecret;

@Repository
public class JdbcTotpRepository implements TotpRepository {

    private final JdbcTemplate jdbcTemplate;
    private final ObjectMapper objectMapper = new ObjectMapper();

    private final RowMapper<TotpSecret> ROW_MAPPER = (rs, n) -> {
        List<String> codes;
        try {
            codes = objectMapper.readValue(rs.getString("recovery_codes_hashed"), new TypeReference<List<String>>() {});
        } catch (Exception e) {
            codes = List.of();
        }
        return new TotpSecret(
                rs.getString("user_id"),
                rs.getString("secret_encrypted"),
                toInstant(rs.getTimestamp("enabled_at")),
                codes,
                toInstant(rs.getTimestamp("created_at")),
                toInstant(rs.getTimestamp("updated_at")));
    };

    private static Instant toInstant(Timestamp ts) {
        return ts != null ? ts.toInstant() : null;
    }

    public JdbcTotpRepository(JdbcTemplate jdbcTemplate) {
        this.jdbcTemplate = jdbcTemplate;
    }

    @Override
    public Optional<TotpSecret> findByUserId(String userId) {
        List<TotpSecret> rows = jdbcTemplate.query(
                "SELECT user_id, secret_encrypted, enabled_at, recovery_codes_hashed, created_at, updated_at " +
                        "FROM stats_user_totp_secrets WHERE user_id = ?",
                ROW_MAPPER, userId);
        return rows.stream().findFirst();
    }

    @Override
    public void upsertPending(String userId, String secret, List<String> recoveryCodesHashed) {
        try {
            String codesJson = objectMapper.writeValueAsString(recoveryCodesHashed);
            jdbcTemplate.update(
                    "INSERT INTO stats_user_totp_secrets (user_id, secret_encrypted, recovery_codes_hashed) " +
                            "VALUES (?, ?, CAST(? AS jsonb)) " +
                            "ON CONFLICT (user_id) DO UPDATE " +
                            "SET secret_encrypted = EXCLUDED.secret_encrypted, " +
                            "    recovery_codes_hashed = EXCLUDED.recovery_codes_hashed, " +
                            "    enabled_at = NULL, updated_at = now()",
                    userId, secret, codesJson);
        } catch (Exception e) {
            throw new IllegalStateException("Не удалось сохранить TOTP-секрет", e);
        }
    }

    @Override
    public void enable(String userId, Instant now) {
        jdbcTemplate.update(
                "UPDATE stats_user_totp_secrets SET enabled_at = ?, updated_at = now() WHERE user_id = ?",
                Timestamp.from(now), userId);
    }

    @Override
    public void delete(String userId) {
        jdbcTemplate.update("DELETE FROM stats_user_totp_secrets WHERE user_id = ?", userId);
    }

    @Override
    public void updateRecoveryCodes(String userId, List<String> hashedCodes) {
        try {
            String codesJson = objectMapper.writeValueAsString(hashedCodes);
            jdbcTemplate.update(
                    "UPDATE stats_user_totp_secrets SET recovery_codes_hashed = CAST(? AS jsonb), updated_at = now() " +
                            "WHERE user_id = ?",
                    codesJson, userId);
        } catch (Exception e) {
            throw new IllegalStateException("Не удалось обновить recovery-коды", e);
        }
    }

    @Override
    public void removeRecoveryCode(String userId, String hashedCode) {
        jdbcTemplate.update(
                "UPDATE stats_user_totp_secrets " +
                        "SET recovery_codes_hashed = recovery_codes_hashed - ?::text, updated_at = now() " +
                        "WHERE user_id = ?",
                hashedCode, userId);
    }
}

-- V7: TOTP 2FA + HIBP-кеш
--
-- stats_user_totp_secrets — один секрет на юзера (пока не enabled,

--
-- stats_password_hibp_cache — кеш ответов HIBP k-anonymity (prefix → suffix list).
-- TTL 24 часа, обновляется по расписанию/при первом запросе.

CREATE TABLE IF NOT EXISTS stats_user_totp_secrets (
    user_id TEXT PRIMARY KEY REFERENCES stats_users(id) ON DELETE CASCADE,
    secret_encrypted TEXT NOT NULL,        -- base32 TOTP секрет (в идеале — зашифрован AES, но пока plaintext для простоты)
    enabled_at TIMESTAMPTZ,                -- NULL = ещё не активирован
    recovery_codes_hashed JSONB NOT NULL DEFAULT '[]'::jsonb,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS stats_password_hibp_cache (
    sha1_prefix CHAR(5) PRIMARY KEY,
    suffixes TEXT NOT NULL,                -- "\r\n"-separated "suffix:count"
    fetched_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS stats_password_hibp_cache_fetched_idx
    ON stats_password_hibp_cache (fetched_at);

import { useState, type FormEvent } from 'react';
import { Panel } from '../../shared/ui/Panel';
import { TextField } from '../../shared/ui/TextField';
import { Button } from '../../shared/ui/Button';
import { CodeInput } from '../../shared/ui/CodeInput';
import { requestEmailChange, confirmEmailChange, ApiError } from '../../shared/api/statsAuth';
import styles from './index.module.scss';

const CODE_LENGTH = 6;

export function UpdateEmail({ email, onChanged }: { email: string; onChanged: (newEmail: string) => void }) {
    const [step, setStep] = useState<'form' | 'code'>('form');
    const [currentPassword, setCurrentPassword] = useState('');
    const [newEmail, setNewEmail] = useState('');
    const [code, setCode] = useState('');
    const [error, setError] = useState<string | null>(null);
    const [notice, setNotice] = useState<string | null>(null);
    const [pending, setPending] = useState(false);

    async function onRequest(e: FormEvent) {
        e.preventDefault();
        setError(null);
        setPending(true);
        try {
            await requestEmailChange(currentPassword, newEmail);
            setStep('code');
        } catch (err) {
            setError(err instanceof ApiError ? err.message : 'Something went wrong.');
        } finally {
            setPending(false);
        }
    }

    async function onConfirm(e: FormEvent) {
        e.preventDefault();
        setError(null);
        setPending(true);
        try {
            const result = await confirmEmailChange(code);
            onChanged(result.user.email);
            setStep('form');
            setCurrentPassword('');
            setNewEmail('');
            setCode('');
            setNotice('Email updated.');
        } catch (err) {
            setError(err instanceof ApiError ? err.message : 'Something went wrong.');
        } finally {
            setPending(false);
        }
    }

    return (
        <Panel title="Change email">
            {error && <p className={styles.error}>{error}</p>}
            {notice && step === 'form' && <p className={styles.notice}>{notice}</p>}
            {step === 'form' ? (
                <form className={styles.form} onSubmit={onRequest}>
                    <TextField
                        id="email-pass"
                        label="Current password"
                        type="password"
                        required
                        value={currentPassword}
                        onChange={e => setCurrentPassword(e.target.value)}
                    />
                    <TextField
                        id="email-new"
                        label="New email"
                        type="email"
                        placeholder={email}
                        required
                        value={newEmail}
                        onChange={e => setNewEmail(e.target.value)}
                    />
                    <Button type="submit" variant="flat" fullWidth disabled={pending}>
                        {pending ? 'Sending code…' : 'Send confirmation code'}
                    </Button>
                </form>
            ) : (
                <form className={styles.form} onSubmit={onConfirm}>
                    <p className={styles.hint}>
                        We sent a 6-digit code to <strong>{newEmail}</strong>.
                    </p>
                    <CodeInput length={CODE_LENGTH} value={code} onChange={setCode} disabled={pending} autoFocus />
                    <Button type="submit" variant="flat" fullWidth disabled={pending || code.length < CODE_LENGTH}>
                        {pending ? 'Confirming…' : 'Confirm new email'}
                    </Button>
                    <button type="button" className={styles.linkBtn} onClick={() => setStep('form')}>
                        Cancel
                    </button>
                </form>
            )}
        </Panel>
    );
}

            setError(err instanceof ApiError ? err.message : 'Something went wrong.');
        } finally {
            setPending(false);

    }

    return (
        <Panel title="Change password">

import { useRef, useState } from 'react';
import { uploadAvatar } from '../../../shared/api/statsAuth';
import styles from './index.module.scss';

interface ProfileAvatarProps {
    currentUrl: string | null;
    email: string;
    onUploaded: (url: string) => void;
}

export function ProfileAvatar({ currentUrl, email, onUploaded }: ProfileAvatarProps) {
    const [loading, setLoading] = useState(false);
    const fileInputRef = useRef<HTMLInputElement>(null);

    const handleFileChange = async (e: React.ChangeEvent<HTMLInputElement>) => {
        const file = e.target.files?.[0];
        if (!file) return;

        if (file.size > 2 * 1024 * 1024) {
            alert('Файл слишком большой (макс. 2MB)');
            return;
        }

        setLoading(true);
        try {
            const { avatarUrl } = await uploadAvatar(file);
            onUploaded(avatarUrl);
        } catch {
            // ИСПРАВЛЕНО: убрана переменная err, чтобы ESLint не ругался
            alert('Не удалось загрузить аватар');
        } finally {
            setLoading(false);
        }
    };

    return (
        <div className={styles.container}>
            <div className={styles.avatarWrapper}>
                {currentUrl ? (
                    <img src={currentUrl} alt="Avatar" className={styles.image} />
                ) : (
                    <div className={styles.placeholder}>{email[0].toUpperCase()}</div>
                )}
                <div className={styles.overlay} onClick={() => fileInputRef.current?.click()}>
                    {loading ? 'Загрузка...' : 'Изменить'}
                </div>
            </div>
            <input
                type="file"
                accept="image/*"
                ref={fileInputRef}
                onChange={handleFileChange}
                style={{ display: 'none' }}
            />
        </div>
    );
}

import type { AnchorHTMLAttributes, ButtonHTMLAttributes, ReactNode } from 'react';
import styles from './index.module.scss';

type Variant = 'primary' | 'ghost' | 'flat';
type Size = 'md' | 'sm';

interface CommonProps {
    variant?: Variant;
    size?: Size;
    fullWidth?: boolean;
    children: ReactNode;
}

type ButtonProps = CommonProps &
    ButtonHTMLAttributes<HTMLButtonElement> & {
        href?: undefined;
    };

type LinkProps = CommonProps &
    AnchorHTMLAttributes<HTMLAnchorElement> & {
        href: string;
    };

// Кнопка-ссылка (OAuth-редирект на /auth/riot/login) или обычная <button> —
// один и тот же паттерн, что в frontend/shared/ui/Button: разная семантика
// тега, одна визуальная оболочка, вместо двух почти одинаковых компонентов.
export function Button({
    variant = 'primary',
    size = 'md',
    fullWidth,
    children,
    className,
    href,
    ...rest
}: ButtonProps | LinkProps) {
    const classes = [styles.base, styles[variant], styles[size], fullWidth ? styles.fullWidth : '', className]
        .filter(Boolean)
        .join(' ');

    if (href !== undefined) {
        return (
            <a className={classes} href={href} {...(rest as AnchorHTMLAttributes<HTMLAnchorElement>)}>
                {children}
            </a>
        );
    }

    return (
        <button className={classes} {...(rest as ButtonHTMLAttributes<HTMLButtonElement>)}>
            {children}
        </button>
    );
}

package tech.extree.stats.adapter.in.web;

import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.Map;
import org.springframework.core.io.FileSystemResource;
import org.springframework.core.io.Resource;
import org.springframework.http.HttpHeaders;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.multipart.MultipartFile;
import tech.extree.stats.adapter.in.web.dto.AccountResponse;
import tech.extree.stats.adapter.in.web.dto.ErrorResponse;
import tech.extree.stats.application.port.in.GetAccountUseCase;
import tech.extree.stats.application.service.AvatarService;
import tech.extree.stats.domain.User;

@RestController
public class AccountController {

    private static final String SESSION_COOKIE = "extree_stats_session";

    private final GetAccountUseCase getAccountUseCase;
    private final AvatarService avatarService;

    public AccountController(GetAccountUseCase getAccountUseCase, AvatarService avatarService) {
        this.getAccountUseCase = getAccountUseCase;
        this.avatarService = avatarService;
    }

    @GetMapping("/api/account/me")
    public ResponseEntity<?> getAccount(@CookieValue(name = SESSION_COOKIE, required = false) String cookie) {
        if (cookie == null) {
            return ResponseEntity.ok(AccountResponse.empty());
        }

        var userOpt = getAccountUseCase.getAccount(cookie);
        if (userOpt.isEmpty()) {
            // Сессия валидна, но пользователя нет в БД или он забанен
            // (AccountService.getAccount() возвращает empty() для забаненных).
            // Отдаём 200 с флагом banned — фронт покажет "аккаунт забанен",
            // а не редирект на /login.
            return ResponseEntity.ok(new AccountResponse(null, true));
        }

        return ResponseEntity.ok(AccountResponse.of(toResponse(userOpt.get())));
    }

    @PostMapping("/api/account/avatar")
    public ResponseEntity<?> uploadAvatar(
            @CookieValue(name = SESSION_COOKIE, required = false) String cookie,
            @RequestParam("file") MultipartFile file) {
        return avatarService.uploadAvatar(cookie, file)
                .<ResponseEntity<?>>map(url -> ResponseEntity.ok(Map.of("avatarUrl", url)))
                .orElseGet(() -> ResponseEntity.badRequest().body(new ErrorResponse("invalid_file")));
    }

    @GetMapping("/api/account/avatar/{filename}")
    public ResponseEntity<Resource> serveFile(@PathVariable String filename) {
        Path baseDir = Paths.get("uploads/avatars").toAbsolutePath().normalize();
        Path file = baseDir.resolve(filename).normalize();
        if (!file.startsWith(baseDir)) {
            return ResponseEntity.notFound().build();
        }
        Resource resource = new FileSystemResource(file);
        if (!resource.exists()) {
            return ResponseEntity.notFound().build();
        }
        return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_TYPE, "image/" + filename.substring(filename.lastIndexOf(".") + 1))
                .body(resource);
    }

    private AccountResponse.AccountUser toResponse(User user) {
        return new AccountResponse.AccountUser(
                user.email(),
                user.emailVerified(),
                user.riotLinked(),
                user.name(),
                user.nickname(),
                getAccountUseCase.isAdmin(user),
                user.avatarUrl(),
                user.isVerified()
        );
    }
}

package tech.extree.stats.adapter.out.persistence;

import java.util.List;
import java.util.UUID;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.core.RowMapper;
import org.springframework.stereotype.Repository;
import tech.extree.stats.application.port.out.LoginEventRepository;
import tech.extree.stats.domain.LoginEvent;

/**
 * Та же таблица/SQL, что ../../stats-backend/loginEvents.js (Node),
 * включая обрезку истории до MAX_EVENTS_PER_USER записей на пользователя.
 */
@Repository
public class JdbcLoginEventRepository implements LoginEventRepository {

    private static final int MAX_EVENTS_PER_USER = 20;

    private static final RowMapper<LoginEvent> ROW_MAPPER =
            (rs, rowNum) -> new LoginEvent(
                    rs.getString("id"),
                    rs.getString("ip"),
                    rs.getString("country"),
                    rs.getString("city"),
                    rs.getTimestamp("created_at").toInstant());

    private final JdbcTemplate jdbcTemplate;

    public JdbcLoginEventRepository(JdbcTemplate jdbcTemplate) {
        this.jdbcTemplate = jdbcTemplate;
    }

    @Override
    public void recordLogin(String userId, String ip, String country, String city) {
        jdbcTemplate.update(
                "INSERT INTO stats_login_events (id, user_id, ip, country, city) VALUES (?, ?, ?, ?, ?)",
                UUID.randomUUID().toString(),
                userId,
                ip,
                country,
                city);
        jdbcTemplate.update(
                "DELETE FROM stats_login_events "
                        + "WHERE user_id = ? AND id NOT IN ("
                        + "  SELECT id FROM stats_login_events WHERE user_id = ? ORDER BY created_at DESC LIMIT ?"
                        + ")",
                userId,
                userId,
                MAX_EVENTS_PER_USER);
    }

    @Override
    public List<LoginEvent> listRecent(String userId, int limit) {
        return jdbcTemplate.query(
                "SELECT id, ip, country, city, created_at FROM stats_login_events "
                        + "WHERE user_id = ? ORDER BY created_at DESC LIMIT ?",
                ROW_MAPPER,
                userId,
                limit);
    }
}
