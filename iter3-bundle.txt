import { useCallback, useEffect, useState } from 'react';
import { Outlet, useNavigate, useParams } from 'react-router-dom';
import { DashboardShell } from '../dashboard-shell';
import { AdminTicketsList } from '../admin-tickets-list';
import {
    getAccount,
    getAdminTickets,
    ApiError,
    type AccountUser,
    type TicketStatus,
    type TicketSummary,
} from '../../shared/api/statsAuth';
import styles from './index.module.scss';

// SRP: этот компонент — только композиция (shell + две колонки + fetch
// списка + навигация по выбору). Сама левая колонка — AdminTicketsList.
// Сама правая — Outlet (AdminTicketPage / AdminTicketsPlaceholder).
//
// Про роутер: useParams/useNavigate, но не про API деталей тикета.
// Про API: только list (getAdminTickets) + account. Детали тикета тянет
// AdminTicketPage сам, в своей зоне ответственности.
export function AdminTicketsLayout() {
    const navigate = useNavigate();
    const params = useParams<{ number?: string }>();
    const activeNumber = params.number ? Number(params.number) : null;

    const [account, setAccount] = useState<AccountUser | null>(null);
    const [checkingAccount, setCheckingAccount] = useState(true);
    const [status, setStatus] = useState<'' | TicketStatus>('');
    const [tickets, setTickets] = useState<TicketSummary[] | null>(null);
    const [error, setError] = useState<string | null>(null);

    useEffect(() => {
        let cancelled = false;
        getAccount().then(({ user, banned }) => {
            if (cancelled) return;
            if (banned || !user) {
                navigate('/login');
                return;
            }
            setAccount(user);
            setCheckingAccount(false);
        });
        return () => {
            cancelled = true;
        };
    }, [navigate]);

    useEffect(() => {
        if (checkingAccount) return;
        let cancelled = false;
        setTickets(null);
        setError(null);
        getAdminTickets(status || undefined)
            .then(result => {
                if (!cancelled) setTickets(result.tickets);
            })
            .catch(err => {
                if (cancelled) return;
                setError(err instanceof ApiError ? err.message : 'Something went wrong — please try again.');
            });
        return () => {
            cancelled = true;
        };
    }, [checkingAccount, status]);

    const handleSelect = useCallback(
        (number: number) => {
            navigate(`/admin/tickets/${number}`);
        },
        [navigate]
    );

    const dashUser = account ? { email: account.email, isAdmin: account.isAdmin } : null;

    return (
        <DashboardShell user={dashUser} title="" wide>
            <div className={styles.layout}>
                <aside className={styles.list}>
                    {checkingAccount ? (
                        <div className={styles.center}>Loading…</div>
                    ) : error ? (
                        <div className={styles.center}>
                            <p className={styles.error}>{error}</p>
                        </div>
                    ) : (
                        <AdminTicketsList
                            tickets={tickets}
                            activeNumber={activeNumber}
                            onSelect={handleSelect}
                            status={status}
                            onStatusChange={setStatus}
                        />
                    )}
                </aside>
                <main className={styles.detail}>
                    <Outlet />
                </main>
            </div>
        </DashboardShell>
    );
}

import { useMemo, useState } from 'react';
import { formatRelativeTime } from '../../shared/lib/formatRelativeTime';
import { type TicketStatus, type TicketSummary } from '../../shared/api/statsAuth';
import styles from './index.module.scss';

const CATEGORY_LABELS: Record<string, string> = {
    account: 'Account & login',
    bug: 'Bug report',
    billing: 'Billing',
    other: 'Other',
};

const STATUS_FILTERS: Array<{ key: '' | TicketStatus; label: string }> = [
    { key: '', label: 'All' },
    { key: 'open', label: 'Open' },
    { key: 'in_progress', label: 'In progress' },
    { key: 'closed', label: 'Closed' },
];

interface AdminTicketsListProps {
    tickets: TicketSummary[] | null;
    activeNumber: number | null;
    onSelect: (number: number) => void;
    status: '' | TicketStatus;
    onStatusChange: (status: '' | TicketStatus) => void;
}

// SRP: левая колонка — только фильтры, поиск, карточки. Про роутер и
// бэк не знает. Активный тикет — акцентная левая полоса + фон, без
// "нативной рамки" (как в первой итерации).
export function AdminTicketsList({ tickets, activeNumber, onSelect, status, onStatusChange }: AdminTicketsListProps) {
    const [search, setSearch] = useState('');

    const counts = useMemo(() => {
        if (!tickets) return { total: 0, open: 0, in_progress: 0, closed: 0 };
        return {
            total: tickets.length,
            open: tickets.filter(t => t.status === 'open').length,
            in_progress: tickets.filter(t => t.status === 'in_progress').length,
            closed: tickets.filter(t => t.status === 'closed').length,
        };
    }, [tickets]);

    const visible = useMemo(() => {
        if (!tickets) return null;
        const q = search.trim().toLowerCase();
        if (!q) return tickets;
        return tickets.filter(t => t.subject.toLowerCase().includes(q) || t.email.toLowerCase().includes(q));
    }, [tickets, search]);

    function countFor(key: '' | TicketStatus): number {
        if (key === '') return counts.total;
        return counts[key];
    }

    return (
        <div className={styles.container}>
            <div className={styles.header}>
                <div className={styles.titleRow}>
                    <h1 className={styles.title}>Inbox</h1>
                    {tickets && <span className={styles.badge}>{counts.total}</span>}
                </div>

                <div className={styles.searchWrapper}>
                    <svg
                        className={styles.searchIcon}
                        viewBox="0 0 24 24"
                        fill="none"
                        stroke="currentColor"
                        strokeWidth="2"
                        strokeLinecap="round"
                        strokeLinejoin="round"
                        aria-hidden="true"
                    >
                        <circle cx="11" cy="11" r="8" />
                        <line x1="21" y1="21" x2="16.65" y2="16.65" />
                    </svg>
                    <input
                        type="search"
                        className={styles.searchInput}
                        placeholder="Search by subject or email…"
                        value={search}
                        onChange={e => setSearch(e.target.value)}
                    />
                </div>

                <nav className={styles.filters} aria-label="Filter tickets by status">
                    {STATUS_FILTERS.map(f => (
                        <button
                            key={f.key || 'all'}
                            type="button"
                            className={`${styles.filterBtn} ${status === f.key ? styles.filterBtnActive : ''}`}
                            onClick={() => onStatusChange(f.key)}
                        >
                            <span>{f.label}</span>
                            <span className={styles.filterCount}>{countFor(f.key)}</span>
                        </button>
                    ))}
                </nav>
            </div>

            <div className={styles.scrollArea}>
                {tickets === null && <div className={styles.placeholder}>Loading…</div>}
                {visible && visible.length === 0 && (
                    <div className={styles.placeholder}>
                        {search.trim() ? 'No tickets match your search.' : 'No tickets in this filter.'}
                    </div>
                )}
                {visible && visible.length > 0 && (
                    <ul className={styles.list}>
                        {visible.map(t => (
                            <li key={t.number}>
                                <button
                                    type="button"
                                    className={`${styles.card} ${activeNumber === t.number ? styles.cardActive : ''}`}
                                    onClick={() => onSelect(t.number)}
                                >
                                    <div className={styles.cardTop}>
                                        <StatusDot status={t.status} />
                                        <span className={styles.number}>#{t.number}</span>
                                        <span className={styles.time}>{formatRelativeTime(t.time)}</span>
                                    </div>
                                    <div className={styles.subject}>{t.subject}</div>
                                    <div className={styles.meta}>
                                        <span className={styles.email}>{t.email}</span>
                                        <span className={styles.category}>
                                            {CATEGORY_LABELS[t.category] ?? t.category}
                                        </span>
                                    </div>
                                </button>
                            </li>
                        ))}
                    </ul>
                )}
            </div>
        </div>
    );
}

function StatusDot({ status }: { status: TicketStatus }) {
    return <span className={`${styles.dot} ${styles[`dot_${status}`]}`} aria-hidden="true" />;
}

import { Avatar } from '../../shared/ui/Avatar';
import { formatRelativeTime } from '../../shared/lib/formatRelativeTime';
import styles from './index.module.scss';

interface ThreadMessage {
    id: string;
    isSupport: boolean;
    authorLabel: string;
    message: string;
    time: string;
}

interface TicketThreadProps {
    messages: ThreadMessage[];
}

// Comment-style thread (Linear / GitHub Issues): без пузырей, только
// Avatar + author + relative time + свободный текст. Меньше визуального
// шума, легче читается длинная переписка. Используется и в админке,
// и в публичной странице тикета.
export function TicketThread({ messages }: TicketThreadProps) {
    return (
        <div className={styles.thread}>
            {messages.map(m => (
                <article key={m.id} className={`${styles.comment} ${m.isSupport ? styles.commentSupport : ''}`}>
                    <Avatar label={m.authorLabel} isSupport={m.isSupport} />
                    <div className={styles.body}>
                        <header className={styles.head}>
                            <span className={styles.author}>{m.authorLabel}</span>
                            <span className={styles.time}>{formatRelativeTime(m.time)}</span>
                        </header>
                        <div className={styles.message}>{m.message}</div>
                    </div>
                </article>
            ))}
        </div>
    );
}

import { useEffect, useState, type FormEvent } from 'react';
import { useNavigate, useParams } from 'react-router-dom';
import { StatusPill } from '../../shared/ui/StatusPill';
import { TextArea } from '../../shared/ui/TextArea';
import { Button } from '../../shared/ui/Button';
import { IconButtonGroup } from '../../shared/ui/IconButtonGroup';
import { EmptyState } from '../../shared/ui/EmptyState';
import { TicketThread } from '../../widgets/ticket-thread';
import { CannedTemplatePicker } from '../../widgets/canned-responses';
import { formatRelativeTime } from '../../shared/lib/formatRelativeTime';
import {
    getAdminTicket,
    replyToTicketAsAdmin,
    setTicketStatus,
    verifyUser,
    type AdminTicketDetail,
    type TicketReply,
    type TicketStatus,
} from '../../shared/api/statsAuth';
import styles from './index.module.scss';

const STATUS_OPTIONS = [
    { value: 'open', label: 'Open' },
    { value: 'in_progress', label: 'In progress' },
    { value: 'closed', label: 'Closed' },
] as const;

export function AdminTicketPage() {
    const { number } = useParams<{ number: string }>();
    const navigate = useNavigate();
    const [ticket, setTicket] = useState<AdminTicketDetail | null>(null);
    const [replies, setReplies] = useState<TicketReply[]>([]);
    const [message, setMessage] = useState('');
    const [pending, setPending] = useState(false);
    const [isVerifying, setIsVerifying] = useState(false);
    const [loadError, setLoadError] = useState<string | null>(null);

    useEffect(() => {
        if (!number) return;
        let cancelled = false;
        setLoadError(null);
        getAdminTicket(number)
            .then(res => {
                if (cancelled) return;
                setTicket(res.ticket);
                setReplies(res.replies);
            })
            .catch(() => {
                if (!cancelled) setLoadError('Failed to load ticket.');
            });
        return () => {
            cancelled = true;
        };
    }, [number]);

    async function onReply(e: FormEvent) {
        e.preventDefault();
        if (!number || !message.trim()) return;
        setPending(true);
        try {
            await replyToTicketAsAdmin(number, message);
            setMessage('');
            const res = await getAdminTicket(number);
            setTicket(res.ticket);
            setReplies(res.replies);
        } catch {
            alert('Failed to send reply');
        } finally {
            setPending(false);
        }
    }

    async function onStatusChange(newStatus: TicketStatus) {
        if (!number) return;
        try {
            await setTicketStatus(number, newStatus);
            setTicket(t => (t ? { ...t, status: newStatus } : null));
        } catch {
            alert('Failed to update status');
        }
    }

    async function handleVerify() {
        if (!ticket || !ticket.userId) return;
        setIsVerifying(true);
        try {
            const newStatus = !ticket.isUserVerified;
            await verifyUser(ticket.userId, newStatus);
            setTicket(t => (t ? { ...t, isUserVerified: newStatus } : null));
        } catch {
            alert('Failed to update verification status');
        } finally {
            setIsVerifying(false);
        }
    }

    if (loadError) {
        return (
            <div className={styles.center}>
                <p className={styles.errorText}>{loadError}</p>
            </div>
        );
    }

    if (!ticket) {
        return <div className={styles.center}>Loading…</div>;
    }

    const threadMessages = [
        { id: 'original', isSupport: false, authorLabel: 'User', message: ticket.message, time: ticket.time },
        ...replies.map(r => ({
            id: r.id,
            isSupport: r.author === 'admin',
            authorLabel: r.author === 'admin' ? 'Support (You)' : 'User',
            message: r.message,
            time: r.time,
        })),
    ];

    const statusIndex = STATUS_OPTIONS.findIndex(o => o.value === ticket.status);

    return (
        <div className={styles.page}>
            <header className={styles.header}>
                <button
                    type="button"
                    className={styles.backBtn}
                    onClick={() => navigate('/admin/tickets')}
                    aria-label="Back to tickets"
                >
                    ←
                </button>
                <div className={styles.headerMain}>
                    <div className={styles.headerTop}>
                        <span className={styles.number}>#{ticket.number}</span>
                        <StatusPill status={ticket.status} />
                    </div>
                    <h1 className={styles.title}>{ticket.subject}</h1>
                    <div className={styles.meta}>
                        <span className={styles.email}>{ticket.email}</span>
                        <span className={styles.sep}>·</span>
                        <span className={styles.category}>{ticket.category}</span>
                        <span className={styles.sep}>·</span>
                        <span className={styles.time}>opened {formatRelativeTime(ticket.time)}</span>
                    </div>
                </div>
                <div className={styles.headerActions}>
                    <IconButtonGroup
                        aria-label="Ticket actions"
                        items={[
                            {
                                key: 'open',

                    <TicketThread messages={threadMessages} />
                </section>

                {ticket.status !== 'closed' ? (

                        <TextArea
                            id="reply"
                            label="Reply"
                            placeholder="Type your reply here…"
                            required
                            value={message}

                        </div>

                        title="This ticket is closed"
                        description="Reopen it to send a new reply."
                        action={
                            <Button variant="ghost" size="sm" onClick={() => onStatusChange('open')}>
                                Reopen ticket
                            </Button>
                        }
                    />
                )}
            </div>
        </div>
    );
}

// SRP: только форматирование даты. Используется везде, где бэкенд отдаёт
// ISO-строку или "Wed, 21 Oct 2026 13:57:38 GMT" (UtcTime.format в Java).
// Возвращает human-friendly "2h ago" / "Yesterday" / "Oct 21".
// Один формат на весь продукт — никаких "GMT" в UI.

const MINUTE = 60 * 1000;
const HOUR = 60 * MINUTE;
const DAY = 24 * HOUR;

export function formatRelativeTime(input: string | Date | null | undefined): string {
    if (!input) return '';
    const date = input instanceof Date ? input : parseDate(input);
    if (!date || isNaN(date.getTime())) return '';

    const now = Date.now();
    const diff = now - date.getTime();

    if (diff < MINUTE) return 'just now';
    if (diff < HOUR) {
        const m = Math.floor(diff / MINUTE);
        return `${m}m ago`;
    }
    if (diff < DAY) {
        const h = Math.floor(diff / HOUR);
        return `${h}h ago`;
    }
    if (diff < 2 * DAY) return 'Yesterday';
    if (diff < 7 * DAY) {
        const d = Math.floor(diff / DAY);
        return `${d}d ago`;
    }

    // Для старых — компактная дата: "Oct 21" или "Oct 21, 2025" если не этот год
    const nowDate = new Date(now);
    const sameYear = date.getFullYear() === nowDate.getFullYear();
    return date.toLocaleDateString('en-US', {
        month: 'short',
        day: 'numeric',
        year: sameYear ? undefined : 'numeric',
    });
}

function parseDate(input: string): Date | null {
    // "Wed, 21 Oct 2026 13:57:38 GMT" — формат UtcTime.format (Java).
    // JS Date.parse не всегда ест, поэтому пробуем сами.
    const direct = new Date(input);
    if (!isNaN(direct.getTime())) return direct;

    // Дополнительный парсинг RFC 1123 с "GMT" суффиксом
    const match = input.match(/^\w{3},\s+(\d{1,2})\s+(\w{3})\s+(\d{4})\s+(\d{2}):(\d{2}):(\d{2})\s+GMT$/);
    if (match) {
        const [, day, mon, year, hh, mm, ss] = match;
        const months: Record<string, number> = {
            Jan: 0,
            Feb: 1,
            Mar: 2,
            Apr: 3,
            May: 4,
            Jun: 5,
            Jul: 6,
            Aug: 7,
            Sep: 8,
            Oct: 9,
            Nov: 10,
            Dec: 11,
        };
        return new Date(Date.UTC(Number(year), months[mon] ?? 0, Number(day), Number(hh), Number(mm), Number(ss)));
    }
    return null;
}

import { useEffect, useRef, useState } from 'react';
import { getCannedResponses, type CannedResponse } from '../../shared/api/statsAuth';
import styles from './index.module.scss';

interface CannedTemplatePickerProps {
    onPick: (body: string) => void;
    disabled?: boolean;
}

// SRP: компонент только выбирает шаблон и отдаёт его через onPick.
// CRUD шаблонов (создание/удаление) — отдельная страница в следующей
// итерации. Здесь же — только "вставить существующий", как в Zendesk/
// Intercom (кнопка Insert template ▾ над textarea).
export function CannedTemplatePicker({ onPick, disabled }: CannedTemplatePickerProps) {
    const [open, setOpen] = useState(false);
    const [templates, setTemplates] = useState<CannedResponse[] | null>(null);
    const [loadError, setLoadError] = useState<string | null>(null);
    const containerRef = useRef<HTMLDivElement>(null);

    useEffect(() => {
        if (!open || templates !== null) return;
        getCannedResponses()
            .then(result => setTemplates(result.responses))
            .catch(() => setLoadError('Failed to load templates'));
    }, [open, templates]);

    useEffect(() => {
        function handleClickOutside(event: MouseEvent) {
            if (containerRef.current && !containerRef.current.contains(event.target as Node)) {
                setOpen(false);
            }
        }
        if (open) {
            document.addEventListener('mousedown', handleClickOutside);
            return () => document.removeEventListener('mousedown', handleClickOutside);
        }
    }, [open]);

    function handlePick(body: string) {
        onPick(body);
        setOpen(false);
    }

    return (
        <div className={styles.wrapper} ref={containerRef}>
            <button
                type="button"
                className={styles.trigger}
                onClick={() => setOpen(v => !v)}
                disabled={disabled}
                aria-haspopup="listbox"
                aria-expanded={open}
            >
                <span>Insert template</span>
                <svg
                    className={`${styles.caret} ${open ? styles.caretOpen : ''}`}
                    viewBox="0 0 24 24"
                    fill="none"
                    stroke="currentColor"
                    strokeWidth="2"
                    strokeLinecap="round"
                    strokeLinejoin="round"
                >
                    <polyline points="6 9 12 15 18 9" />
                </svg>
            </button>

            {open && (
                <div className={styles.menu} role="listbox">
                    {loadError && <div className={styles.error}>{loadError}</div>}
                    {templates === null && !loadError && <div className={styles.empty}>Loading…</div>}
                    {templates?.length === 0 && <div className={styles.empty}>No templates yet.</div>}
                    {templates && templates.length > 0 && (
                        <ul className={styles.list}>
                            {templates.map(t => (
                                <li key={t.id}>
                                    <button type="button" className={styles.item} onClick={() => handlePick(t.body)}>
                                        <span className={styles.itemTitle}>{t.title}</span>
                                        <span className={styles.itemBody}>{t.body}</span>
                                    </button>
                                </li>
                            ))}
                        </ul>
                    )}
                </div>
            )}
        </div>
    );
}

    AnchorHTMLAttributes<HTMLAnchorElement> & {
        href: string;
    };

// Кнопка-ссылка (OAuth-редирект на /auth/riot/login) или обычная <button> —
// один и тот же паттерн, что в frontend/shared/ui/Button: разная семантика
// тега, одна визуальная оболочка, вместо двух почти одинаковых компонентов.
export function Button({
    variant = 'primary',
    size = 'md',
    fullWidth,
    children,
    className,
    href,
    ...rest
}: ButtonProps | LinkProps) {
    const classes = [styles.base, styles[variant], styles[size], fullWidth ? styles.fullWidth : '', className]
        .filter(Boolean)
        .join(' ');

    if (href !== undefined) {
        return (
            <a className={classes} href={href} {...(rest as AnchorHTMLAttributes<HTMLAnchorElement>)}>
                {children}
            </a>
        );
    }

    return (
        <button className={classes} {...(rest as ButtonHTMLAttributes<HTMLButtonElement>)}>
            {children}
        </button>
    );
}

async function postJson<T>(path: string, body: unknown): Promise<T> {
    const res = await fetch(path, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        credentials: 'same-origin',
        body: JSON.stringify(body),
    });
    const data = await res.json().catch(() => ({}));
    if (!res.ok) {
        const code = typeof data.error === 'string' ? data.error : 'internal_error';
        throw new ApiError(code, messageFor(code));
    }
    return data as T;
}

    }
    return data as T;
}
